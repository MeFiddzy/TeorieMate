\newcommand{\rightangletick}[4][0.15cm]{%
    \coordinate (tickA) at ($(#2)!#1!(#3)$);
    \coordinate (tickB) at ($(#2)!#1!(#4)$);
    \coordinate (tickC) at ($(tickA)+(tickB)-(#2)$);
    \draw (tickA) -- (tickC) -- (tickB);
}

\newcommand{\regularPolygon}[5]{%
    \def\previtem{}%
    \def\firstitem{}%
    \foreach \x [count=\i] in #2 {
        \coordinate (\x) at ($#4 + (\i*360/#1 + #3:#5)$);
        \ifnum\i=1
            \xdef\firstitem{\x}%
        \else
            \draw (\previtem) -- (\x);
        \fi
        \xdef\previtem{\x}%
    }
    \draw (\previtem) -- (\firstitem);
}
