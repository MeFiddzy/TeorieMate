\newcommand{\rightangletick}[4][0.15cm]{%
    \coordinate (tickA) at ($(#2)!#1!(#3)$);
    \coordinate (tickB) at ($(#2)!#1!(#4)$);
    \coordinate (tickC) at ($(tickA)+(tickB)-(#2)$);
    \draw (tickA) -- (tickC) -- (tickB);
}

\newcommand{\regularPolygon}[5]{% #1=count, #2=list, #3=shift, #4=center, #5=rad
    \def\previtem{}%
    \def\firstitem{}%
    \foreach \x [count=\i] in #2 {
        \coordinate (\x) at ($#4 + (\i*360/#1 + #3:#5)$);
        \ifnum\i=1
            \xdef\firstitem{\x}%
        \else
            \draw (\previtem) -- (\x);
        \fi
        \xdef\previtem{\x}%
    }
    \draw (\previtem) -- (\firstitem);
}

\newcommand{\point}[2]{%
    \draw[fill] (#1) circle(1px) node[#2]{#1};
}

\newcommand{\markline}[4][]{%
    \path (#2); \pgfgetlastxy{\ax}{\ay}
    \path (#3); \pgfgetlastxy{\bx}{\by}

    \draw[#1] (#2) arc[start angle=180, end angle=0, 
                    x radius={0.5*sqrt((\bx-\ax)^2+(\by-\ay)^2)/1cm}, 
                    y radius=#4];
}
