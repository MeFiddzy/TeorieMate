\ptitle{Principalele mulțimi de numere}
    \psubtitle{Mulțimea numerelor naturale ($\mathbb{N}$)}
        Mulțimea numerelor naturale este formate din numerele mai mare sau egale cu 0 și fără o parte decimală.
        \[ \mathbb{N} = \{0, 1, 2, 3, ..., \infty\} \]
        Definim și mulțimea numerelor naturale nenule ($\mathbb{N}^*$) în următorul fel.
        \[ \mathbb{N}^* = \mathbb{N} \setminus \{0\} = \{1, 2, 3, ..., \infty\} \]
    \psubtitle{Mulțimea numerelor întregi ($\mathbb{Z}$)}
        Mulțimea numerelor întregi este formată din toate numerele fără o parte decimală sau prin reuniunea dintre mulțimea numerelor naturale cu opusul ei.
        \[ \mathbb{Z} = \{ -\infty, ..., -2, -1, 0, 1, 2, ..., \infty \} = \mathbb{N} \cup -\mathbb{N}\]
        Mai definim și mulțimea numerelor întregi negative ($\mathbb{Z}_-$) și pozitive ($\mathbb{Z}_+$).
        \[ \mathbb{Z}_- = \mathbb{Z} \setminus \mathbb{N}, \qquad \mathbb{Z}_+ = \mathbb{N}^* \]
        Reuniune celor doua rezultă în mulțimea numerelor întregi nenule ($\mathbb{Z}^*$).
        \[ \mathbb{Z}^* = \mathbb{Z}_- \cup \mathbb{Z}_+ = \mathbb{Z} \setminus \{0\} \] 
    \psubtitle{Mulțimea numerelor raționale ($\mathbb{Q}$)}
        Mulțimea numerelor raționale este formată din toate numerele care se pot forma prin împărțirea a două numere întregi.
        \[ \mathbb{Q} = \left\{ \left. \frac{a}{b} \right| a,b \in \mathbb{Z}, b \neq 0 \right\} \]
        Definim și mulțimea numerelor raționale negative ($\mathbb{Q}_-$) și pozitive ($\mathbb{Q}_+$).
        \[ \mathbb{Q}_- = \left\{ \left. -\frac{a}{b}\right| a, b \in \mathbb{N}^* \right\}, \qquad 
        \mathbb{Q}_+ = \mathbb{Q} \setminus \mathbb{Q}_- \setminus \{0\} \]
        Reuniune celor două rezultă în mulțimea numerelor raționale nenule ($\mathbb{Q}^*$).
        \[ \mathbb{Q}^* = \mathbb{Q}_- \cup \mathbb{Q}_+ = \mathbb{Q} \setminus \{0\} \]
    \psubtitle{Mulțimea numerelor iraționale ($\mathbb{R} \setminus \mathbb{Q}$)}
        Mulțimea numerelor iraționale este formată din toate numerele care nu pot fii scrise ca o fracție dintre două numere întregi. \\
        Exemple de numere iraționale.
        \[ \{ \pm \sqrt{n}, \pi, \phi \} \subset \mathbb{R} \setminus \mathbb{Q}, n \not - \text{pp.}\] 
        Exemple de numere \textbf{raționale}, dar unele presoane cred că ele sunt iraționale.
        \[ \{ 0,(9); 0,(3) \} \subset \mathbb{Q} \not \subset \mathbb{R} \]
    \psubtitle{Mulțimea numerelor reale ($\mathbb{R}$)}
        Mulțimea numerelor reale este cea mai mare mulțime învățata, ea este formată prin reuniunea mulțimii numerelor iraționale ($\mathbb{R} \setminus \mathbb{Q}$) cu mulțimea numerelor raționale ($\mathbb{Q}$).
        \[ \mathbb{R} = (\mathbb{R} \setminus \mathbb{Q}) \cup \mathbb{Q} \]
        Definim mulțimile numerelor reale pozitive ($\mathbb{R}_+$) și negative ($\mathbb{R}_-$) în felul următor.
        \[ \mathbb{R}_+ = \{ x | x \in \mathbb{R}, x > 0 \}, \qquad \mathbb{R}_- = \mathbb{R} \setminus \mathbb{R}_+ \setminus \{0\} \]
    \psubtitle{Relația de incluziune dintre principalele mulțimi de numere}
        Mulțiimile principale sunt incluse una în următoarea în ordine în care le-am definit (excluzând mulțimea numerelor iraționale, care este inclusa doar în mulțimea numerelor reale).
        \[ \mathbb{N} \subset \mathbb{Z} \subset \mathbb{Q} \subset \mathbb{R}, \quad (\mathbb{R} \setminus \mathbb{Q}) \subset \mathbb{R} \]
        \textbf{Notă}: Relația de inclusiune \textbf{nu} înseamnă că cardinalele mulțimilor sunt într-o relație de inegalitate, deoarece mulțimile nu sunt finite, ci sunt infinite.
        \[
        \left. 
        \begin{array}{l}
            A \subset B \\
            A, B - \text{infinite}
        \end{array}
        \right\} \centernot\Longrightarrow C(A) < C(B) 
        \]

\ptitle{Intervale de numere reale ($\mathbb{R}$)}
    \psubtitle{Definirea intervalului}
    Un \textbf{interval} este o parte a numerelor reale. \\
    Un interval poate fi:
    \begin{enumerate}
        \item mărginit (ambele limite sunt finite ($\neq \pm \infty$))
        \item nemărginit (una dintre limite (inferioară sau/și superioară) este infinită ($= \pm \infty$))
    \end{enumerate}
    \psubtitle{Intervalul mărginit}
        Următoarele identități definesc intervalul mărginit.
        \begin{enumerate}
            \item $ \{x \in \mathbb{R} | a < x < b\} = (a;b),\ \forall a < b; a, b \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | a \le x \le b\} = [a; b],\ \forall a < b; a, b \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | a < x \le b\} = (a; b],\ \forall a < b; a, b \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | a \le x < b\} = [a; b),\ \forall a < b; a, b \in \mathbb{R} $
        \end{enumerate}
        Exemple de intervale mărginite.
        \begin{enumerate}
            \item $ \{x \in \mathbb{R} | 1 < x \le 3\} = (1; 3] $
            \item $ \{x \in \mathbb{R} | 3 \le x < \sqrt{10}\} = [3; \sqrt{10}) $
            \item $ \left\{x \in \mathbb{R} | -\frac{1}{2} \le x \le 2\right\} = \left[-\frac{1}{2}; 2\right] $
            \item $ \{x \in \mathbb{R} | -\sqrt{3} < x < 10\} = (-\sqrt{3}; 10) $
            \item $ \{x \in \mathbb{R} | 2 < x \le \sqrt{3}\} = \emptyset $
        \end{enumerate}
    \psubtitle{Intervalul nemărginit}
        Următoarele identități definesc intervalul nemărginit.
        \begin{enumerate}
            \item $ \{x \in \mathbb{R} | a < x\} = (a; +\infty), \forall a \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | a \le x\} = [a; +\infty), \forall a \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | x < a\} = (-\infty; a), \forall a \in \mathbb{R} $
            \item $ \{x \in \mathbb{R} | x \le a\} = (-\infty; a], \forall a \in \mathbb{R} $
        \end{enumerate}
        \newpage
        Exemple de intervale nemărginite.
        \begin{enumerate}
            \item $ \{x \in \mathbb{R} | \sqrt{2} < x\} = (\sqrt{2}; +\infty) $
            \item $ \{x \in \mathbb{R} | x \ge -1\} = [-1; +\infty) $
            \item $ \{x \in \mathbb{R} | x < \frac{3}{4}\} = (-\infty; \frac{3}{5}) $
            \item $ \{x \in \mathbb{R} | x \le \pi\} = (-\infty; \pi] $
            \item $ \mathbb{R} = (-\infty; +\infty) $
        \end{enumerate}
    \psubtitle{Identități suplimentare}
        Următoarele identități sunt folosite în exercițiile cu intervale.
        \begin{enumerate}
            \item $ \{ x \in \mathbb{R} | |x| < a\} = (-a; a),\ \forall a \in \mathbb{R}_+ $
            \item $ \{ x \in \mathbb{R} | |x| \le a\} = [-a; a],\ \forall a \in \mathbb{R}_+ $
            \item $ \{ x \in \mathbb{R} | |x| > a\} = \mathbb{R} \setminus (-a; a),\ \forall a \in \mathbb{R}_+ $
            \item $ \{ x \in \mathbb{R} | |x| \ge a\} = \mathbb{R} \setminus [-a; a],\ \forall a \in \mathbb{R}_+ $
        \end{enumerate}

    \ptitle{Operații cu intervale}
        \psubtitle{Reuniunea ($\cup$)}
            Reuniune dintre două mulțimi este o altă mulțime, ea fiind formată din toate elementele prezente cel puțin una dintre
            mulțimile inițiale.
            \[ A \cup B = \{ x | x \in A \text{ sau } x \in B \} \]
            Example de reuniuni de intervale.
            \begin{enumerate}
                \item $(-3; 2] \cup (1; 8) = (-3; 8)$
                \item $(-\infty; 10] \cup [11; +\infty) = \mathbb{R} \setminus (10; 11)$
                \item $(9; 10] \cup (10; 20) = (9; 20)$
            \end{enumerate}
        \psubtitle{Intersecția ($\cap$)}
            Intersecția a două mulțimi este o altă mulțime, ea fiind formată din toate elementele care sunt prezente in amble mulțimi
            inițiale.
            \[ A \cup B = \{ x | x \in A \text{ și } x \in B \} \]
            Example de intersecții de intervale.
            \begin{enumerate}
                \item $(-3; 2] \cap (1; 8) = (1; 2]$
                \item $(-\infty; 10] \cap [11; +\infty) = \emptyset$
                \item $(9; 10] \cap (10; 20) = \{10\}$
            \end{enumerate}
        \psubtitle{Diferența ($\setminus$)}
            Intersecția a două mulțimi este o altă mulțime, ea fiind formată din toate elementele care aparțin primei mulțimi, dar nu în a doua.
            \[ A \setminus B = \{ x | x \in A \text{ și } x \not \in B \} \]
            \newpage
            Example de diferențe de intervale.
            \begin{enumerate}
                \item $(-3; 2] \setminus (1; 8) = (-3; 1]$
                \item $(-\infty; 10] \cap [2; +\infty) = (-\infty, 2)$
                \item $(1; 2] \cap (-1; 7) = \emptyset$
            \end{enumerate}
